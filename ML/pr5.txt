import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import dendrogram, linkage
df = pd.read_csv("sales_data_sample.csv", encoding="latin1")

print(df.head())
print(df.shape)
print(df.isnull().sum())
features = ['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'MSRP']

X = df[features].copy()

X = X.fillna(X.median())

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

print("Data preprocessing completed")
wcss = []

for i in range(1, 11):
    kmeans = KMeans(n_clusters=i, random_state=42, n_init=10)
    kmeans.fit(X_scaled)
    wcss.append(kmeans.inertia_)

plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.grid()
plt.show()
k = 3

kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)

df['KMeans_Cluster'] = kmeans.fit_predict(X_scaled)

print(df[['SALES', 'KMeans_Cluster']].head())

print("\nCluster Counts:")
print(df['KMeans_Cluster'].value_counts())
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 5))

plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=df['KMeans_Cluster'],
    cmap='viridis'
)

plt.title("K-Means Clustering")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.show()
hc = AgglomerativeClustering(
    n_clusters=k,
    linkage='ward'
)

df['Hierarchical_Cluster'] = hc.fit_predict(X_scaled)

print(df[['SALES', 'Hierarchical_Cluster']].head())

print("\nCluster Counts:")
print(df['Hierarchical_Cluster'].value_counts())
plt.figure(figsize=(10, 5))

Z = linkage(X_scaled[:200], method='ward')

dendrogram(Z)

plt.title("Hierarchical Clustering Dendrogram")
plt.xlabel("Data Points")
plt.ylabel("Euclidean Distance")
plt.show()
plt.figure(figsize=(8, 5))

plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=df['Hierarchical_Cluster'],
    cmap='rainbow'
)

plt.title("Hierarchical Clustering")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.show()
print("K-Means Clustering:")
print(df['KMeans_Cluster'].value_counts().sort_index())

print("\nHierarchical Clustering:")
print(df['Hierarchical_Cluster'].value_counts().sort_index())

print("\nAverage Sales by K-Means Cluster:")
print(df.groupby('KMeans_Cluster')['SALES'].mean())

print("\nAverage Sales by Hierarchical Cluster:")
print(df.groupby('Hierarchical_Cluster')['SALES'].mean())








Q1. What is the aim of this practical?

To perform customer or sales data clustering using K-Means and Hierarchical Clustering algorithms.

Q2. Which type of Machine Learning is used?

Unsupervised Machine Learning because the dataset does not contain predefined output labels.

Q3. What is clustering?

Clustering is the process of grouping similar data points into clusters based on their similarities.

Q4. Which algorithms are used?
1. K-Means Clustering.
2. Agglomerative (Hierarchical) Clustering.
Q5. What is K-Means Clustering?

K-Means is an unsupervised learning algorithm that divides data into K clusters based on the distance from cluster centroids.

Q6. What is Hierarchical Clustering?

It is an unsupervised learning algorithm that creates a hierarchy of clusters by combining or dividing data points.

Q7. What is the difference between K-Means and Hierarchical Clustering?
K-Means	Hierarchical
Uses centroids	Uses hierarchy of clusters
Requires K value	Can build hierarchy before choosing clusters
Faster for large datasets	Can be computationally expensive
Uses cluster assignment	Uses dendrogram

Q8. What is StandardScaler?

It standardizes features so that they have a mean of 0 and standard deviation of 1.

Q9. What is the Elbow Method?

The Elbow Method is used to find the optimal number of clusters by plotting WCSS against the number of clusters.

Q10. What is WCSS?

WCSS stands for Within-Cluster Sum of Squares. It measures the sum of squared distances between data points and their cluster centroids.

Q11. What is PCA?

PCA stands for Principal Component Analysis. It reduces the number of dimensions while retaining as much information as possible.

Q12. What is a dendrogram?

A dendrogram is a tree-like diagram that shows how clusters are merged in Hierarchical Clustering.










Q13. What is Agglomerative Clustering?

It is a bottom-up approach in which each data point starts as an individual cluster, and similar clusters are gradually merged.

Q14. What is linkage='ward'?

Ward linkage merges clusters in a way that minimizes the increase in within-cluster variance.

Q15. Why do we use two clustering algorithms?

To compare K-Means and Hierarchical Clustering results and understand how both algorithms group sales data.
