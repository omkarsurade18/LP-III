pragma solidity ^0.8.34;

// Smart contract for a simple bank
contract SimpleBank {

    address public owner;   // Stores the owner's address
    uint256 public balance; // Stores the bank balance

    // Constructor executes only once during deployment
    constructor()
    {
        owner = msg.sender; // Set deployer as owner
        balance = 0;        // Initialize balance to zero
    }

    // Function to deposit money
    function deposit(uint256 amount) public {

        // Only owner can deposit money
        require(msg.sender == owner, "Only the owner can deposit money");

        balance += amount; // Add amount to balance
    }

    // Function to withdraw money
    function withdraw(uint256 amount) public {

        // Only owner can withdraw money
        require(msg.sender == owner, "Only the owner can withdraw money");

        // Check whether sufficient balance is available
        require(amount <= balance, "Insufficient balance");

        balance -= amount; // Subtract amount from balance
    }

    // Function to check current balance
    function getBalance() public view returns (uint256) {

        return balance; // Return current balance
    }
}







The program creates a smart contract named SimpleBank.



The owner variable stores the Ethereum address of the contract owner.



The balance variable stores the current bank balance.



The constructor executes only once when the contract is deployed.



The constructor assigns the deployer's address as the owner.



Initially, the bank balance is set to zero.



The deposit function is used to add money to the balance.



Only the owner can deposit money.



The withdraw function is used to withdraw money from the balance.



Only the owner can withdraw money.



Before withdrawal, the program checks whether sufficient balance is available.



The getBalance function displays the current balance.



The view keyword ensures that the function does not modify contract data.



The require statement checks conditions and stops execution if a condition is false.
