pragma solidity ^0.8.20;

contract StudentData {

    // Structure to store student details
    struct Student {
        uint id;
        string name;
        uint marks;
    }

    // Array to store multiple students
    Student[] public students;

    // Add student data
    function addStudent(uint id, string memory name, uint marks) public {
        students.push(Student(id, name, marks));
    }

    // Display student data
    function getStudent(uint index) public view returns (uint, string memory, uint) {
        Student memory s = students[index];
        return (s.id, s.name, s.marks);
    }

    // Return total number of students
    function getTotalStudents() public view returns (uint) {
        return students.length;
    }
}




1. The MIT license specifies the license under which the contract is shared.

2. Solidity version 0.8.20 is specified.

3. A smart contract named StudentData is created.

4. A structure named Student is created to store student details.

5. The structure contains three fields: student ID, name, and marks.

6. A public array named students is created to store multiple student records.

7. The addStudent() function adds a new student's ID, name, and marks to the array.

8. The getStudent() function retrieves the details of a student using their array index.

9. The getTotalStudents() function returns the total number of students stored in the array.

10. The contract is compiled and deployed using Remix IDE.

11. After deployment, we can execute the functions to add and display student records.

12. All student records are stored on the Ethereum blockchain.
